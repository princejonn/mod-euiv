assassinate_ruler = {
	category = covert
	require_acceptance = no
	is_visible = {
		ai = no
		government = monarchy
		FROM = {
			government = monarchy
			ai = yes
		}
		full_idea_group = spy_ideas
	}
	is_allowed = {
		employed_advisor = {
			type = spymaster
		}
		has_spy_network_in = {
			who = FROM
			value = 95
		}
		dip_power = 250
		treasury = 500
	}
	on_accept = {
		add_spy_network_in = {
			who = FROM
			value = -95
		}
		add_dip_power = -250
		add_treasury = -500
		FROM = {
			kill_ruler = yes
			add_stability = -1
		}
	}
	on_decline = {
	}
	ai_will_do = {
		always = no
	}
}

forge_claim_on_throne = {
	category = dynasty
	require_acceptance = no
	is_visible = {
		ai = no
		government = monarchy
		FROM = {
			government = monarchy
		}
		full_idea_group = spy_ideas
	}
	is_allowed = {
		employed_advisor = {
			type = spymaster
		}
		has_spy_network_in = {
			who = FROM
			value = 95
		}
		dip_power = 250
		heir_claim = 50
		marriage_with = FROM
		treasury = 500
		FROM = {
			dynasty = ROOT
		}
		custom_trigger_tooltip = {
			tooltip = WEAK_HEIR_CLAIM_TT
			FROM = {
				OR = {
					has_heir = no
					AND = {
						has_heir = yes
						NOT = {
							heir_claim = 50
						}
					}
				}
			}
		}
		custom_trigger_tooltip = {
			tooltip = ROOT_HAS_MORE_PRESTIGE_THAN_FROM_TT
			AND = {
				prestige = FROM
				NOT = {
					FROM = {
						prestige = ROOT
					}
				}
			}
		}
	}
	on_accept = {
		add_dip_power = -250
		add_treasury = -500
		add_prestige = -25
		add_spy_network_in = {
			who = FROM
			value = -95
		}
		add_casus_belli = {
			type = cb_restore_personal_union
			target = FROM
			months = 300
		}
	}
	on_decline = {
	}
	ai_will_do = {
		always = no
	}
}

slander_heir_claim = {
	category = covert
	require_acceptance = no
	is_visible = {
		ai = no
		government = monarchy
		FROM = {
			government = monarchy
		}
		full_idea_group = spy_ideas
	}
	is_allowed = {
		employed_advisor = {
			type = spymaster
		}
		has_spy_network_in = {
			who = FROM
			value = 75
		}
		dip_power = 75
		treasury = 75
		FROM = {
			heir_claim = 10
		}
	}
	on_accept = {
		add_dip_power = -75
		add_treasury = -500
		add_spy_network_in = {
			who = FROM
			value = -75
		}
		if = {
			limit = {
				FROM = {
					heir_claim = 50
				}
			}
			FROM = {
				add_heir_claim = -50
			}
		}
		else_if = {
			limit = {
				FROM = {
					heir_claim = 40
				}
			}
			FROM = {
				add_heir_claim = -40
			}
		}
		else_if = {
			limit = {
				FROM = {
					heir_claim = 30
				}
			}
			FROM = {
				add_heir_claim = -30
			}
		}
		else_if = {
			limit = {
				FROM = {
					heir_claim = 20
				}
			}
			FROM = {
				add_heir_claim = -20
			}
		}
		else_if = {
			limit = {
				FROM = {
					heir_claim = 10
				}
			}
			FROM = {
				add_heir_claim = -10
			}
		}
	}
	on_decline = {
	}
	ai_will_do = {
		always = no
	}
}
